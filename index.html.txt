<script>
let assets=JSON.parse(localStorage.getItem('pi_assets')||'[]');
let profile=JSON.parse(localStorage.getItem('pi_profile')||'{}');
let extra=JSON.parse(localStorage.getItem('pi_extra')||'{}');
let charts={};

function show(id,btn){
 document.querySelectorAll('main section').forEach(s=>s.classList.remove('active'));
 document.getElementById(id).classList.add('active');
 document.querySelectorAll('nav button').forEach(b=>b.classList.remove('active'));
 btn.classList.add('active');
 if(id==='dashboard'||id==='riesgo') refresh();
 if(id==='pendientes') missing();
}
function currency(){
 return document.getElementById('currency')?.value||'CLP';
}
function money(v){
 return new Intl.NumberFormat('es-CL',{style:'currency',currency:currency(),
 maximumFractionDigits:0}).format(v||0);
}
function pct(v){return (v||0).toFixed(2)+'%'}
function save(){
 localStorage.setItem('pi_assets',JSON.stringify(assets));
}
function addAsset(){
 let ticker=document.getElementById('ticker').value.trim().toUpperCase();
 let name=document.getElementById('assetName').value.trim();
 let value=Number(document.getElementById('value').value);
 if(!ticker||!name||!Number.isFinite(value)||value<=0){
 alert('Ingresa ticker, nombre y un valor actual mayor que cero.');return;
 }
 assets.push({
 id:Date.now(),ticker,name,
 category:document.getElementById('category').value,
 sector:document.getElementById('sector').value||'No informado',
 region:document.getElementById('region').value||'No informado',
 currency:document.getElementById('assetCurrency').value,
 value,performance:Number(document.getElementById('performance').value)||0,
 fee:Number(document.getElementById('fee').value)||0,
 benchmark:document.getElementById('benchmark').value,
 account:document.getElementById('account').value,
 leverage:document.getElementById('leverage').value
 });
 save();clearForm();render();refresh();
}
function clearForm(){
 ['ticker','assetName','sector','region','value','performance','fee',
 'benchmark','account'].forEach(id=>document.getElementById(id).value='');
}
function removeAsset(id){
 assets=assets.filter(a=>a.id!==id);save();render();refresh();
}
function total(){return assets.reduce((s,a)=>s+a.value,0)}
function weight(a){return total()?a.value/total()*100:0}
function render(){
 document.getElementById('assetRows').innerHTML=assets.map(a=>`
 <tr><td><b>${esc(a.ticker)}</b></td><td>${esc(a.name)}</td>
 <td>${esc(a.category)}</td><td>${money(a.value)}</td>
 <td>${pct(weight(a))}</td><td>${pct(a.performance)}</td>
 <td><button class="danger" onclick="removeAsset(${a.id})">×</button></td></tr>
 `).join('');
}
function esc(s){
 return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function groupBy(field){
 let o={};assets.forEach(a=>{let k=a[field]||'No informado';o[k]=(o[k]||0)+weight(a)});
 return o;
}
function draw(id,type,labels,data){
 if(charts[id])charts[id].destroy();
 let canvas=document.getElementById(id);
 if(!canvas)return;
 charts[id]=new Chart(canvas,{
 type,data:{labels,datasets:[{data,label:'Peso (%)',
 backgroundColor:['#16866b','#315b85','#d5a348','#9878bc',
 '#dc7777','#65a9bd','#8b9b83','#b4a0a0','#66768a'],
 borderWidth:1}]},
 options:{responsive:true,maintainAspectRatio:false,
 plugins:{legend:{position:type==='bar'?'bottom':'right'}},
 scales:type==='bar'?{y:{beginAtZero:true}}:{}}
 });
}
function refresh(){
 let t=total();
 let weightedReturn=t?assets.reduce((s,a)=>s+a.value*a.performance/100,0)/t*100:0;
 let gain=t?assets.reduce((s,a)=>s+a.value*(a.performance/(100+a.performance)),0):0;
 document.getElementById('total').textContent=money(t);
 document.getElementById('return').textContent=pct(weightedReturn);
 document.getElementById('gain').textContent=money(gain);
 document.getElementById('count').textContent=assets.length;
 draw('allocationChart','doughnut',Object.keys(groupBy('category')),Object.values(groupBy('category')));
 draw('performanceChart','bar',assets.map(a=>a.ticker),assets.map(a=>a.performance));
 draw('geoChart','doughnut',Object.keys(groupBy('region')),Object.values(groupBy('region')));
 let sorted=[...assets].sort((a,b)=>b.value-a.value);
 document.getElementById('largest').textContent=pct(sorted[0]?weight(sorted[0]):0);
 let hhi=assets.reduce((s,a)=>s+Math.pow(weight(a),2),0);
 document.getElementById('hhi').textContent=hhi.toFixed(0);
 document.getElementById('leveraged').textContent=assets.filter(a=>a.leverage==='yes').length;
 let fees=assets.reduce((s,a)=>s+a.value*a.fee,0);
 document.getElementById('weightedFee').textContent=t?pct(fees/t):'N/D';
 draw('sectorChart','doughnut',Object.keys(groupBy('sector')),Object.values(groupBy('sector')));
 draw('geoRiskChart','doughnut',Object.keys(groupBy('region')),Object.values(groupBy('region')));
 let top=sorted[0];
 document.getElementById('executive').innerHTML=assets.length?
 `<div class="alert good">Se registraron ${assets.length} posiciones, con un valor total de ${money(t)}.</div>
 <div class="alert">La mayor posición es ${esc(top.ticker)}, con ${pct(weight(top))} del portafolio.</div>
 <div class="alert">Rentabilidad ponderada informada: ${pct(weightedReturn)}. Este indicador no reemplaza la TIR ni la rentabilidad ponderada por tiempo.</div>`
 :'No hay activos registrados.';
 document.getElementById('riskAlerts').innerHTML=assets.length?
 alertsHTML():'Incorpora activos para calcular alertas.';
 document.getElementById('advancedIndicators').innerHTML=
 '<div class="alert">Volatilidad, Sharpe, beta, correlación y máximo drawdown requieren series históricas de precios, benchmark y fechas. No se deben inferir a partir de la rentabilidad acumulada.</div>';
}
function alertsHTML(){
 let out=[];
 let top=[...assets].sort((a,b)=>weight(b)-weight(a))[0];
 if(top&&weight(top)>25)out.push(['bad','Concentración individual: '+top.ticker+' representa '+pct(weight(top))+'.']);
 let sectors=groupBy('sector');
 Object.entries(sectors).forEach(([s,v])=>{
 if(s!=='No informado'&&v>30)out.push(['bad','Concentración sectorial: '+s+' representa '+pct(v)+'.']);
 });
 let lev=assets.filter(a=>a.leverage==='yes');
 if(lev.length)out.push(['bad','Se identificaron instrumentos apalancados. Revisar objetivo diario, costos y riesgo de pérdida.']);
 let fees=assets.filter(a=>a.fee>1);
 if(fees.length)out.push(['','Hay instrumentos con costos anuales declarados superiores al 1%. Verificar TER y comisiones.']);
 if(!out.length)out.push(['good','No se detectaron alertas básicas de concentración con los umbrales utilizados. Esto no significa que la cartera sea de bajo riesgo.']);
 return out.map(x=>`<div class="alert ${x[0]}">${x[1]}</div>`).join('');
}

function analyze(){
 refresh();
 let r=[];
 if(!assets.length)r.push('No hay activos. Registra instrumentos antes de emitir un diagnóstico.');
 let h=Number(profile.horizon||0),tol=Number(profile.tolerance||3),
 cap=Number(profile.capacity||3);
 if(h&&h<5&&assets.some(a=>a.category.includes('acciones')||a.category==='Acciones'))
 r.push('Horizonte corto: revisar si la exposición a renta variable es compatible con las necesidades de liquidez.');
 if(tol!==cap)r.push('Existe una diferencia entre tolerancia psicológica y capacidad financiera. Conviene utilizar la dimensión más restrictiva para definir el riesgo asumible.');
 if(assets.some(a=>a.leverage==='yes'))r.push('Reducir la dependencia de instrumentos apalancados en el núcleo patrimonial.');
 if(assets.filter(a=>a.category.includes('ETF')).length>1)
 r.push('Revisar solapamientos entre ETF y exposición indirecta a las mismas empresas.');
 if(assets.some(a=>weight(a)>20))
 r.push('Evaluar concentración por posición y considerar límites por emisor.');
 if(assets.some(a=>a.performance< -20))
 r.push('Revisar los activos con pérdidas superiores al 20% mediante análisis fundamental; no vender automáticamente por pérdidas.');
 if(assets.some(a=>a.fee>1))
 r.push('Comparar costos de los instrumentos con alternativas equivalentes.');
 if(profile.debt==='yes')r.push('Priorizar evaluación de deuda de alto costo antes de incrementar riesgo de inversión.');
 if(Number(profile.emergency||0)<3)
 r.push('Revisar fondo de emergencia y liquidez antes de asumir mayor exposición a activos volátiles.');
 if(!r.length)r.push('No hay alertas automáticas adicionales. Se necesita información histórica y contextual para un diagnóstico completo.');
 document.getElementById('recommendations').innerHTML=r.map((x,i)=>
 `<div class="alert">${i+1}. ${x}</div>`).join('');
 document.getElementById('actionPlan').innerHTML=[
 '1. Completar los datos faltantes y verificar los valores registrados.',
 '2. Revisar solapamientos, concentración y costos.',
 '3. Definir una asignación objetivo compatible con el perfil.',
 '4. Establecer reglas de rebalanceo y revisar periódicamente.'
 ].map(x=>`<div class="alert">${x}</div>`).join('');
}
function project(){
 let initial=Number(document.getElementById('initial').value)||0;
 let monthly=Number(document.getElementById('monthly').value)||0;
 let rate=Number(document.getElementById('rate').value)/100;
 let inf=Number(document.getElementById('inflation').value)/100;
 let years=Math.max(1,Math.min(100,Number(document.getElementById('years').value)||1));
 if(rate<=-1||inf<=-1){alert('Revisa las tasas ingresadas.');return;}
 let labels=[],values=[],capital=initial;
 for(let y=0;y<=years;y++){
  if(y>0){
   let rm=Math.pow(1+rate,1/12)-1;
   for(let m=0;m<12;m++)capital=capital*(1+rm)+monthly;
  }
  labels.push('Año '+y);values.push(Math.round(capital));
 }
 let future=capital,real=future/Math.pow(1+inf,years);
 document.getElementById('futureValue').textContent=money(future);
 document.getElementById('realValue').textContent=money(real);
 document.getElementById('projectionNote').textContent=
 'Simulación nominal con tasa constante, aportes al final de cada mes e inflación constante. No incluye impuestos, comisiones ni variaciones del mercado. No es una predicción.';
 if(charts.projectionChart)charts.projectionChart.destroy();
 charts.projectionChart=new Chart(document.getElementById('projectionChart'),{
 type:'line',data:{labels,datasets:[{label:'Patrimonio estimado',data:values,
 borderColor:'#16866b',backgroundColor:'rgba(22,134,107,.12)',fill:true,tension:.3}]},
 options:{responsive:true,maintainAspectRatio:false}
 });
}
function missing(){
 let checks=[
 ['Perfil del inversionista',!!profile.age&&!!profile.horizon],
 ['País y moneda base',!!profile.country&&!!profile.currency],
 ['Objetivo de inversión',!!profile.goal],
 ['Tolerancia y capacidad de riesgo',!!profile.tolerance&&!!profile.capacity],
 ['Activos y valores actuales',assets.length>0],
 ['Rentabilidad por activo',assets.length>0&&assets.every(a=>a.performance!==null)],
 ['Sector y región',assets.length>0&&assets.every(a=>a.region!=='No informado'&&a.sector!=='No informado')],
 ['Comisiones',assets.length>0&&assets.every(a=>a.fee>0)],
 ['Aportes y liquidez',!!profile.contribution&&!!profile.emergency],
 ['Deudas y restricciones',!!extra.liabilities&&!!extra.restrictions]
 ];
 let n=checks.filter(x=>x[1]).length;
 let p=Math.round(n/checks.length*100);
 document.getElementById('missingList').innerHTML=checks.map(x=>
 `<div style="padding:8px;border-bottom:1px solid #eee">
 ${x[1]?'✓':'○'} ${x[0]} <span class="tag">${x[1]?'Completo':'Pendiente'}</span></div>`
 ).join('');
 document.getElementById('completeness').textContent=p+'%';
 document.getElementById('completeBar').style.width=p+'%';
}

function importData(e){
 let f=e.target.files[0];if(!f)return;
 let reader=new FileReader();
 reader.onload=()=>{
 try{
 let d=JSON.parse(reader.result);
 if(!Array.isArray(d.assets))throw Error();
 assets=d.assets;profile=d.profile||{};extra=d.extra||{};
 save();localStorage.setItem('pi_profile',JSON.stringify(profile));
 localStorage.setItem('pi_extra',JSON.stringify(extra));
 render();refresh();missing();alert('Datos importados.');
 }catch(err){alert('Archivo JSON inválido.');}
 };
 reader.readAsText(f);
}

async function runOCR(){
 if(!images.length){alert('Primero carga una imagen.');return;}
 let out=document.getElementById('ocrText');
 let status=document.getElementById('ocrStatus');
 out.value='';
 for(let i=0;i<images.length;i++){
  status.textContent='Procesando imagen '+(i+1)+' de '+images.length+'...';
  try{
   let result=await Tesseract.recognize(images[i],'eng+spa');
   out.value+='\n--- Imagen '+(i+1)+' ---\n'+result.data.text+'\n';
  }catch(e){out.value+='\nNo se pudo leer imagen '+(i+1);}
 }
 status.textContent='Extracción terminada. Comprueba los datos manualmente.';
}
